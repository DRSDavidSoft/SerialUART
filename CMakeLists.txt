cmake_minimum_required(VERSION 3.16)
project(SerialUART VERSION 2.2.0 LANGUAGES C)
if(APPLE)
  set(CMAKE_INSTALL_RPATH "@loader_path/../lib")
elseif(UNIX)
  set(CMAKE_INSTALL_RPATH "$ORIGIN/../lib")
endif()
include(GNUInstallDirs)
include(CTest)
option(BUILD_SHARED_LIBS "Build library for language bindings" OFF)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(serialuart lib/SerialUART.c)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_sources(serialuart PRIVATE lib/linux_native.c)
endif()
add_library(SerialUART::serialuart ALIAS serialuart)
add_executable(SerialLoopback examples/SerialLoopback.c)
target_link_libraries(SerialLoopback PRIVATE serialuart)
target_include_directories(serialuart PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/lib> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_definitions(serialuart PRIVATE SUART_BUILD)
if(BUILD_SHARED_LIBS)
  target_compile_definitions(serialuart PUBLIC SUART_SHARED)
endif()
set_target_properties(serialuart PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 2)
if(WIN32)
  target_link_libraries(serialuart PRIVATE setupapi advapi32)
elseif(APPLE)
  target_link_libraries(serialuart PRIVATE "-framework IOKit" "-framework CoreFoundation")
endif()
add_library(suart_protocol STATIC examples/protocol.c)
target_include_directories(suart_protocol PUBLIC examples)
add_library(suart_session STATIC cli/session.c cli/options.c cli/devices.c cli/profiles.c cli/keys.c cli/windows_args.c)
target_include_directories(suart_session PUBLIC cli)
target_link_libraries(suart_session PUBLIC serialuart suart_protocol)
if(WIN32)
  target_link_libraries(suart_session PUBLIC shell32)
endif()
add_executable(SendCommand SendCommand.c examples/network.c)
target_link_libraries(SendCommand PRIVATE serialuart suart_protocol suart_session)
if(WIN32)
  target_link_libraries(SendCommand PRIVATE ws2_32 shell32)
endif()
foreach(target serialuart suart_protocol suart_session SendCommand SerialLoopback)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
if(BUILD_TESTING)
  enable_language(CXX)
  add_executable(network_echo tests/network_echo.c examples/network.c)
  target_link_libraries(network_echo PRIVATE serialuart suart_protocol)
  if(WIN32)
    target_link_libraries(network_echo PRIVATE ws2_32)
  endif()
  add_executable(protocol_test tests/protocol_test.c)
  target_link_libraries(protocol_test PRIVATE suart_protocol serialuart)
  add_test(NAME protocol COMMAND protocol_test)
  add_executable(session_test tests/session_test.c)
  target_link_libraries(session_test PRIVATE suart_session)
  add_test(NAME session COMMAND session_test)
  add_executable(session_view tests/session_view.c)
  target_link_libraries(session_view PRIVATE suart_session)
  # Test the real selection policy with a discovery/open fixture, on every OS.
  add_executable(device_policy_test tests/device_policy_test.c cli/devices.c cli/options.c)
  target_include_directories(device_policy_test PRIVATE lib cli examples)
  target_link_libraries(device_policy_test PRIVATE suart_protocol)
  if(MSVC)
    target_compile_options(device_policy_test PRIVATE /utf-8)
    target_compile_definitions(device_policy_test PRIVATE _CRT_SECURE_NO_WARNINGS)
  endif()
  add_test(NAME device_policy COMMAND device_policy_test)
  add_executable(cpp_test tests/cpp_test.cpp)
  target_link_libraries(cpp_test PRIVATE serialuart)
  add_test(NAME cpp_linkage COMMAND cpp_test)
  add_test(NAME help COMMAND SendCommand --help)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME session_features COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/session_features.py $<TARGET_FILE:SendCommand> $<TARGET_FILE:session_view>)
    set_tests_properties(session_features PROPERTIES TIMEOUT 45)
    if(UNIX)
      add_executable(native_test tests/native_test.c)
      target_link_libraries(native_test PRIVATE serialuart)
      if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        add_executable(discovery_test tests/discovery_test.c)
        target_link_libraries(discovery_test PRIVATE serialuart)
        set(SUART_DISCOVERY_TEST $<TARGET_FILE:discovery_test>)
      else()
        set(SUART_DISCOVERY_TEST unused)
      endif()
      add_test(NAME native_serial COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/native_test.py $<TARGET_FILE:native_test> $<TARGET_FILE:SerialLoopback> $<TARGET_FILE:SendCommand> ${SUART_DISCOVERY_TEST})
      set_tests_properties(native_serial PROPERTIES TIMEOUT 45)
    endif()
    add_test(NAME release_validation COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/release_test.py)
    add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:SendCommand> $<TARGET_FILE:network_echo>)
    set_tests_properties(integration PROPERTIES TIMEOUT 90)
  endif()
endif()
install(TARGETS serialuart SendCommand SerialLoopback EXPORT SerialUARTTargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES lib/SerialUART.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT SerialUARTTargets NAMESPACE SerialUART:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SerialUART)
include(CMakePackageConfigHelpers)
find_package(Git QUIET)
set(SUART_SOURCE_COMMIT "unknown")
if(Git_FOUND)
  execute_process(COMMAND ${GIT_EXECUTABLE} rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} OUTPUT_VARIABLE SUART_SOURCE_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
endif()
configure_file(cmake/BUILD-INFO.txt.in ${CMAKE_CURRENT_BINARY_DIR}/BUILD-INFO.txt @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/BUILD-INFO.txt DESTINATION ${CMAKE_INSTALL_DOCDIR})
configure_package_config_file(cmake/SerialUARTConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/SerialUARTConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SerialUART)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/SerialUARTConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/SerialUARTConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/SerialUARTConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/SerialUART)
install(FILES README.md CONTRIBUTORS.md Changelog.txt DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(DIRECTORY docs examples DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(FILES data/SIMPLYSERIAL-LICENSE.txt data/simplyserial-boards.json DESTINATION ${CMAKE_INSTALL_DOCDIR}/data)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_GENERATOR ZIP TGZ)
include(CPack)
